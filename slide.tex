\begin{frame}{Bài toán: sentiment có liên hệ với phản ứng giá?}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{block}{Vấn đề NLP}
Một headline có thể nhắc nhiều doanh nghiệp với sắc thái khác nhau. Sentiment cấp câu dễ gán sai tín hiệu cho từng cổ phiếu.
\end{block}
\begin{block}{Giải pháp ABSA}
FinABSA dự đoán sentiment cho \highlight{một target đã xác định} bằng Target--Other masking.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Câu hỏi kiểm định}
\begin{enumerate}
\item Prediction có ghép đúng article--ticker pair không?
\item Sentiment có liên hệ với abnormal return sau tin không?
\item Kết quả có bền vững qua năm, event window và lag không?
\end{enumerate}
\end{block}
\begin{alertblock}{Nguyên tắc diễn giải}
Đánh giá \textbf{association quan sát được}, không khẳng định sentiment là nguyên nhân gây biến động giá.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Pipeline: entity linking, ABSA và market validation}
\begin{columns}[T,onlytextwidth]

\begin{block}{Model contract}
\small
\textbf{Headline:} ``HPG tăng mạnh, SSI điều chỉnh.''\\[0.08cm]
\textbf{Input HPG:} ``\alert{Target} tăng mạnh, \textcolor{FinGray}{Other} điều chỉnh.''\\[0.08cm]

\centering\small

$\rightarrow$ target masking $\rightarrow$ FinABSA\\
$\rightarrow$ ticker--day aggregation $\rightarrow$ statistical tests
\end{block}

Đơn vị tài chính & ticker--day \\

\end{tabular}
\end{block}
\begin{alertblock}{Hệ quả}
Model không tự dự đoán ticker; entity linking phải được kiểm soát độc lập.

\begin{frame}{Event study: loại trừ biến động thị trường chung}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{block}{Căn chỉnh thời gian}
\small
\begin{itemize}
\item Giá là dữ liệu ngày, không có intraday timestamp đáng tin cậy.
\item Bài báo map sang \highlight{phiên giao dịch đầu tiên sau ngày xuất bản}.
\item $r_{i,t}=100[\log(P_{i,t})-\log(P_{i,t-1})]$.
\end{itemize}
\end{block}
\begin{block}{Market model}
\[
R_{i,t}=\alpha_i+\beta_iR_{m,t}+\varepsilon_{i,t},\qquad
AR_{i,t}=R_{i,t}-\widehat{E}(R_{i,t}).
\]
\small VNINDEX là market proxy; cần ít nhất 20 quan sát pre-event để fit market model.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Ba event windows}
\small
\begin{tabular}{p{0.25\linewidth}p{0.65\linewidth}}
$[0,0]$ & phiên event \\
$[0,+1]$ & event và một phiên sau \\
$[0,+3]$ & event và ba phiên sau \\
\end{tabular}
\end{block}
\begin{block}{Suy luận bổ sung}
\small
\begin{itemize}
\item One-sample $t$-test, sign-flip permutation, bootstrap 95\% CI.
\item News-day control; group contrasts; ticker/date fixed effects.
\item Lag robustness tại 0, 1, 2, 3, 5 và 10 phiên.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\source{Brown and Warner (1985); pipeline implementation.}
\end{frame}

% Slide 12
\begin{frame}{Giới hạn, hướng phát triển và kết luận}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\begin{block}{Limitations}
\small
\begin{itemize}
\item Một tháng October cho mỗi năm; regime dependence còn cao.
\item CafeF là nguồn đơn; news selection không ngẫu nhiên.
\item Entity linking có false positive/negative; class imbalance mạnh.
\item Daily prices không cho intraday timing; panel thưa cần HC1 fallback.
\item Nhiều kiểm định nên nominal $p$-value cần đọc thận trọng.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Future work}
\small
\begin{enumerate}
\item Gold labels độc lập, adjudication, F1/MCC/kappa và error analysis.
\item Lưu logits/probabilities để calibration và uncertainty-aware aggregation.
\item Mở rộng thời gian, broad linking, sector/regime interactions.
\item Pre-register tests, placebo/factor models, multiple-testing correction, out-of-time forecasting.
\end{enumerate}
\end{block}
\end{column}
\end{columns}
\vspace{0.06cm}
\begin{alertblock}{Kết luận khoa học}
\small Pipeline NLP-to-finance đã hoàn chỉnh và tái lập. Một vài period-specific estimates có nominal significance, nhưng dấu và độ lớn không ổn định qua năm, event window và lag. Kết quả không chứng minh causal effect, predictive power ổn định hay trading strategy.
\end{alertblock}
